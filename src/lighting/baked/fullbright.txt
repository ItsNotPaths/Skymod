# SkyMod lighting profile: fullbright
# Dev / low-spec preset. No real lighting: kills the sun, floods flat full ambient, and
# disables shadows, fog and specular. With the passthrough tonemap the screen shows the raw
# authored textures (albedo) — handy for spotting content, and cheap on a weak GPU because the
# cascaded-shadow passes skip entirely (shadow_strength = 0) and the fragment work is trivial.

# Sun — off (ambient does all the lighting)
sun_dir = 0.0, 0.0, 1.0
sun_color = 1.0, 1.0, 1.0
sun_intensity = 0.0

# Ambient — flat, full white, slightly over-driven, with the floor pinned. Skyrim's diffuse is
# authored dark for its dim lighting, so flat 1.0 ambient still reads gloomy; the overdrive plus
# the albedo_lift gamma below pull the midtones up to a readable "fullbright" brightness.
ambient_sky = 1.0, 1.0, 1.0
ambient_ground = 1.0, 1.0, 1.0
ambient_intensity = 1.4
ambient_floor = 1.0

# Fog — off
fog_color = 1.0, 1.0, 1.0
fog_start = 0
fog_end = 400000
fog_height_falloff = 0
fog_density = 0
sky_color = 0.5, 0.6, 0.75

# Tonemap — passthrough so albedo reaches the screen unchanged (no curve to darken it)
tonemap = none
exposure = 1.0
white_point = 1.0
contrast = 1.0
saturation = 1.0
color_filter = 1.0, 1.0, 1.0

# Material response — no spec/normal (cheap + flat). albedo_lift is a gamma on the sampled
# diffuse: <1 lifts the dark-authored darks hard (0.45 ≈ a 0.1 texel → 0.35) without blowing
# out the brights, which is the right brightening curve for Skyrim's dim textures.
albedo_lift = 0.45
spec_scale = 0.0
foliage_spec = 0.0
emissive_scale = 1.0
normal_strength = 0.0

# Shadows — off (skips the CSM passes entirely — the main perf win here)
shadow_strength = 0.0
veg_shadows = off
