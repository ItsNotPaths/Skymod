# Keep every run's log in a timestamped logs/ folder instead of overwriting the

persist_logs = false

# Exterior render distance: the streaming window half-size in cells. The loaded
# square is (2*render_distance + 1)^2 cells around the player (2 = 5x5). Higher
# sees farther but streams/draws more.
render_distance = 2

# Grass draw distance in world units (grass is dense, so much shorter than render
# distance). Cells farther than this draw no grass. 0 disables grass entirely.
grass_distance = 8192

# Distant-object LOD radius in cells: how far Skyrim's prebaked LOD meshes (rocks, buildings,
# mountains) render. Terrain itself is whole-world CDLOD with no distance knob, so this is the
# main "how far can I see stuff" dial. Beyond the full-detail render_distance these cells load
# OBJECTS ONLY (CDLOD draws their terrain), so a large radius is cheap. Objects without a prebaked
# LOD mesh (clutter) drop past render_distance.
object_lod_distance = 24

# Distant-tree-billboard radius in cells: how far Skyrim's prebaked tree LOD billboards (flat
# _lod_flat impostors, one shared per-worldspace atlas) render. Trees are the cheapest far tier, so
# this can reach past object_lod_distance to fill the horizon with tree silhouettes. Defaults to
# object_lod_distance when unset (no change). Cells between object_lod_distance and here load trees
# ONLY (no statics). Larger = more resident cells, so bump it deliberately.
# tree_lod_distance = 48
